def test_should_handle_edge_cases_when_converting_servers():
    # Arrange
    from openhands_cli.acp_impl.utils.mcp import convert_acp_mcp_servers_to_agent_format
    from acp.schema import StdioMcpServer, HttpMcpServer, HttpHeader
    
    # Test 1: Empty server list
    result_empty = convert_acp_mcp_servers_to_agent_format([])
    assert result_empty == {}
    
    # Test 2: Server without env field (HttpMcpServer doesn't have env)
    http_server = HttpMcpServer(
        name="http_no_env",
        url="http://localhost:9000",
        headers=[]
    )
    result_no_env = convert_acp_mcp_servers_to_agent_format([http_server])
    assert "http_no_env" in result_no_env
    assert "env" not in result_no_env["http_no_env"]
    assert result_no_env["http_no_env"]["transport"] == "http"
    
    # Test 3: Multiple servers in batch
    stdio_server = StdioMcpServer(
        name="server1",
        command="cmd1",
        args=["arg1"],
        env=[]
    )
    http_server2 = HttpMcpServer(
        name="server2",
        url="http://localhost:8000",
        headers=[]
    )
    result_batch = convert_acp_mcp_servers_to_agent_format([stdio_server, http_server2])
    assert len(result_batch) == 2
    assert "server1" in result_batch
    assert "server2" in result_batch
    assert result_batch["server1"]["transport"] == "stdio"
    assert result_batch["server2"]["transport"] == "http"

def test_should_preserve_all_server_fields_during_conversion():
    # Arrange
    from openhands_cli.acp_impl.utils.mcp import convert_acp_mcp_servers_to_agent_format
    from acp.schema import StdioMcpServer, EnvVariable
    
    server = StdioMcpServer(
        name="comprehensive_server",
        command="/path/to/command",
        args=["--flag1", "value1", "--flag2"],
        env=[
            EnvVariable(name="ENV1", value="val1"),
            EnvVariable(name="ENV2", value="val2"),
        ]
    )
    
    # Act
    result = convert_acp_mcp_servers_to_agent_format([server])
    
    # Assert
    server_config = result["comprehensive_server"]
    assert server_config["command"] == "/path/to/command"
    assert server_config["args"] == ["--flag1", "value1", "--flag2"]
    assert server_config["env"] == {"ENV1": "val1", "ENV2": "val2"}
    assert server_config["transport"] == "stdio"
    assert "name" not in server_config
    # Verify no extra fields were added
    expected_keys = {"command", "args", "env", "transport"}
    assert set(server_config.keys()) == expected_keys
